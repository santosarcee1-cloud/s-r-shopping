<script>
// Kumuha ng mga elemento
const addButtons = document.querySelectorAll('.add-btn');
const cartItemsContainer = document.getElementById('cart-items');
const totalPriceElement = document.getElementById('total-price');
const checkoutBtn = document.getElementById('checkout-btn');

// Listahan ng mga gamit sa kariton
let cart = [];

// Magdagdag ng kaganapan sa bawat butones
addButtons.forEach(button => {
    button.addEventListener('click', () => {
        const name = button.getAttribute('data-name');
        const priceAttr = button.getAttribute('data-price');
        const price = parseFloat(priceAttr);
        
        // Suriin kung wasto ang presyo
        if (!name || isNaN(price) || price <= 0) {
            console.error('Maling datos ng produkto:', { name, priceAttr });
            return;
        }
        
        // Tingnan kung nandoon na ang item
        const existingItem = cart.find(item => item.name === name);
        if (existingItem) {
            existingItem.quantity += 1;
        } else {
            cart.push({ name, price, quantity: 1 });
        }
        
        updateCartDisplay();
    });
});

// I-update ang pagpapakita ng kariton
function updateCartDisplay() {
    // Linisin muna ang laman
    cartItemsContainer.innerHTML = '';
    
    if (cart.length === 0) {
        cartItemsContainer.innerHTML = '<p class="empty-cart">Wala pang laman ang kariton</p>';
        totalPriceElement.textContent = '0.00';
        return;
    }
    
    let total = 0;
    
    // Ipakita ang bawat item
    cart.forEach((item, index) => {
        const itemTotal = item.price * item.quantity;
        total += itemTotal;
        
        const cartItemDiv = document.createElement('div');
        cartItemDiv.className = 'cart-item';
        cartItemDiv.innerHTML = `
            <div>
                <strong>${item.name}</strong><br>
                ₱${item.price.toFixed(2)} x ${item.quantity} = <strong>₱${itemTotal.toFixed(2)}</strong>
            </div>
            <button class="remove-btn" data-index="${index}">Alisin</button>
        `;
        cartItemsContainer.appendChild(cartItemDiv);
    });
    
    totalPriceElement.textContent = total.toFixed(2);
    
    // Maglagay ng kaganapan sa mga butones na pag-alis
    document.querySelectorAll('.remove-btn').forEach(btn => {
        btn.addEventListener('click', () => {
            const index = parseInt(btn.getAttribute('data-index'));
            if (!isNaN(index)) {
                cart.splice(index, 1);
                updateCartDisplay();
            }
        });
    });
}

// Pag-tsek out
checkoutBtn.addEventListener('click', () => {
    if (cart.length === 0) {
        alert('Walang laman ang iyong kariton!');
        return;
    }
    
    const total = cart.reduce((sum, item) => sum + item.price * item.quantity, 0);
    
    alert(`✅ Salamat sa pag-order!\n\nKabuuang halaga: ₱${total.toFixed(2)}\nMakikipag-ugnayan kami sa inyo para sa paghahatid.`);
    
    // Linisin ang kariton matapos mag-order
    cart = [];
    updateCartDisplay();
});
</script>
